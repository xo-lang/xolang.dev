pub enum ApiErr {
  BadRequest(msg: Str)
  NotFound(id: Int)
}

fn (s: Store) get(id: Int) -> User ! ApiErr
  requires id > 0 else ApiErr.BadRequest(msg: "id must be positive")
{
  s.users.with(fn(m) { m.get(id) }).or_err(ApiErr.NotFound(id: id))?
}

fn main(os: Os) ! {
  var r = http.router()
  r.handle("GET /users/{id:Int}", fn(req) -> http.Response ! ApiErr {
    http.json(200, store.get(req.param_int("id"))?)
  })
  http.server(os.net.listen(":8080")?, r).serve()?
}

Every facet checked

What makes Go productive stays: few keywords, one obvious way, fast builds, static binaries. The bugs tests usually miss move into the compiler.

No nil

Exhaustive match

No data races

Value semantics; sharing only through Mutex and Atomic.
